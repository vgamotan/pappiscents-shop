// Pappi Orders — Telegram alert // Deploy in Supabase: Edge Functions → Deploy a new function → Via editor // Name it exactly: notify-telegram (turn OFF "Enforce JWT verification") // Secrets needed (Edge Functions → Secrets): TELEGRAM_BOT_TOKEN, TELEGRAM_CHAT_ID import { createClient } from "npm:@supabase/supabase-js@2"; const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; const peso = (n: number) => "₱" + Number(n).toLocaleString("en-PH", { minimumFractionDigits: 0 }); const deliveryLabel: Record = { nationwide: "Nationwide (3–5 days)", cebu_express: "Cebu City express (Maxim/Grab/Angkas)", pickup: "Pickup", }; const esc = (s: string) => String(s ?? "").replace(/[<>&]/g, (c) => ({ "<": "<", ">": ">", "&": "&" }[c]!)); Deno.serve(async (req) => { if (req.method === "OPTIONS") return new Response("ok", { headers: cors }); try { const { order_id } = await req.json(); if (!order_id) throw new Error("order_id missing"); const db = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!); // Claim the alert once, so the same order can never be announced twice const { data: order, error } = await db .from("orders") .update({ notified_at: new Date().toISOString() }) .eq("id", order_id) .is("notified_at", null) .select("*, customers(name, phone), order_items(name, line, qty, unit_price)") .single(); if (error || !order) return new Response(JSON.stringify({ ok: true, skipped: true }), { headers: { ...cors, "Content-Type": "application/json" } }); // Returning customer? const { count } = await db.from("orders").select("id", { count: "exact", head: true }).eq("customer_id", order.customer_id); const items = order.order_items.map((i: any) => `• ${i.qty} × ${esc(i.name)} ${esc(i.line)} — ${peso(i.qty * i.unit_price)}`).join("\n"); const text = `🛍 New order ${order.order_no}${(count ?? 1) > 1 ? ` ⭐ repeat buyer (${count} orders)` : ""}\n\n` + `${items}\n• Shipping fee — ${Number(order.shipping_fee) ? peso(order.shipping_fee) : "FREE"}\n\nShould receive: ${peso(order.total)} via ${esc(order.payment_method)}\n\n` + `👤 ${esc(order.customers.name)}\n📞 ${esc(order.customers.phone)}\n` + (order.address ? "" : `🚚 ${deliveryLabel[order.delivery] ?? order.delivery}\n`) + (order.address ? `📍 ${esc(order.address)}\n` : "") + (order.customer_note ? `📝 ${esc(order.customer_note)}\n` : "") + `\nCheck the screenshot, then mark it Paid in your dashboard.`; const token = Deno.env.get("TELEGRAM_BOT_TOKEN")!; const chat_id = Deno.env.get("TELEGRAM_CHAT_ID")!; const tg = (method: string, body: unknown) => fetch(`https://api.telegram.org/bot${token}/${method}`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body), }); let sent = false; if (order.proof_path) { const { data: signed } = await db.storage.from("payment-proofs").createSignedUrl(order.proof_path, 60 * 60); if (signed?.signedUrl && text.length <= 1024) { const r = await tg("sendPhoto", { chat_id, photo: signed.signedUrl, caption: text, parse_mode: "HTML" }); sent = r.ok; } else if (signed?.signedUrl) { await tg("sendPhoto", { chat_id, photo: signed.signedUrl, caption: `Payment screenshot for ${order.order_no}` }); } } if (!sent) await tg("sendMessage", { chat_id, text, parse_mode: "HTML" }); return new Response(JSON.stringify({ ok: true }), { headers: { ...cors, "Content-Type": "application/json" } }); } catch (e) { return new Response(JSON.stringify({ ok: false, error: String(e) }), { status: 400, headers: { ...cors, "Content-Type": "application/json" } }); } });
